    59→def normalize_word(w):
    60→    """Lowercase, strip accents/punctuation, expand tiny numbers — for matching only."""
    61→    w = unicodedata.normalize("NFKD", str(w))
    62→    w = "".join(ch for ch in w if not unicodedata.combining(ch))
    63→    w = w.lower().strip()
    64→    w = re.sub(r"[^a-z0-9' ]", "", w)
    65→    w = w.replace("'", "")
    66→    if w in DIGIT_WORDS:
    67→        w = DIGIT_WORDS[w]
    68→    return w

    71→TAG_RE = re.compile(r"^\[([^\[\]]+)\]\s*[:\-–—]?\s*$")

    74→def parse_lyrics(text):
    75→    """Lyric sheet -> list of entries: {'kind': 'tag'|'sing', 'text', 'words'}."""
    76→    entries = []
    77→    for raw in (text or "").splitlines():
    78→        line = raw.strip()
    79→        if not line:
    80→            continue
    81→        m = TAG_RE.match(line)
    82→        if m:
    83→            entries.append({"kind": "tag", "text": m.group(1).strip(), "words": []})
    84→        else:
    85→            entries.append(
    86→                {"kind": "sing", "text": line, "words": [w for w in line.split() if w]}
    87→            )
    88→    return entries

    91→def fmt_clock(t, comma=False, force_hours=False):
    92→    """MM:SS.mmm (or H:MM:SS.mmm) — integer-ms math so 59.9996 never rounds to '60.000'."""
    93→    total_ms = int(round(max(0.0, float(t)) * 1000))
    94→    h, rem = divmod(total_ms, 3_600_000)
    95→    m, rem = divmod(rem, 60_000)
    96→    s, ms = divmod(rem, 1000)
    97→    sep = "," if comma else "."
    98→    if h or force_hours:
    99→        return f"{h:02d}:{m:02d}:{s:02d}{sep}{ms:03d}"
   100→    return f"{m:02d}:{s:02d}{sep}{ms:03d}"
   101→
   102→
   103→def fmt_yt(t):
   104→    total_s = int(round(max(0.0, float(t))))
   105→    h, rem = divmod(total_s, 3600)
   106→    m, s = divmod(rem, 60)
   107→    if h:
   108→        return f"{h}:{m:02d}:{s:02d}"
   109→    return f"{m}:{s:02d}"
   110→
   111→
   112→def fmt_lrc_stamp(t):
   113→    """[MM:SS.xx] — centisecond math so 59.996 s becomes [01:00.00], never [00:60.00]."""
   114→    total_cs = int(round(max(0.0, float(t)) * 100))
   115→    m, rem = divmod(total_cs, 6000)
   116→    s, cs = divmod(rem, 100)
   117→    return f"[{m:02d}:{s:02d}.{cs:02d}]"

   120→def fmt_plain(t):
   121→    return fmt_yt(t)

   166→def _resample_linear(x, sr_in, sr_out):
   167→    if sr_in == sr_out:
   168→        return x
   169→    n_out = int(round(len(x) * sr_out / sr_in))
   170→    t_in = np.arange(len(x)) / sr_in
   171→    t_out = np.arange(n_out) / sr_out
   172→    return np.interp(t_out, t_in, x).astype(np.float32)

   175→def load_audio_16k(path):
   176→    """Decode any common audio format to float32 mono 16 kHz.
   177→
   178→    soundfile handles WAV/FLAC/MP3/OGG; plain av.open handles M4A/AAC and
   179→    anything else (no keyword args, so it works on both old and new PyAV).
   180→    """
   181→    try:
   182→        import soundfile as sf
   183→
   184→        data, sr = sf.read(path, dtype="float32", always_2d=False)
   185→        if data.ndim > 1:
   186→            data = data.mean(axis=1)
   187→        if sr != 16000:
   188→            data = _resample_linear(data, sr, 16000)
   189→        return np.asarray(data, dtype=np.float32)
   190→    except Exception:
   191→        pass
   192→    try:
   193→        import av
   194→
   195→        chunks = []
   196→        with av.open(path, mode="r") as container:
   197→            streams = [s for s in container.streams if s.type == "audio"]
   198→            if not streams:
   199→                raise RuntimeError("no audio stream")
   200→            resampler = av.AudioResampler(format="s16", layout="mono", rate=16000)
   201→            for frame in container.decode(streams[0]):
   202→                out = resampler.resample(frame)
   203→                if out is None:
   204→                    continue
   205→                frames = out if isinstance(out, (list, tuple)) else [out]
   206→                for rf in frames:
   207→                    chunks.append(np.asarray(rf.to_ndarray()).reshape(-1))
   208→        if not chunks:
   209→            raise RuntimeError("no decoded frames")
   210→        return (np.concatenate(chunks).astype(np.float32) / 32768.0).astype(np.float32)
   211→    except Exception as exc:
   212→        raise RuntimeError(f"could not decode audio {path!r}: {exc}")

   215→def transcribe_words(audio_path, model_size, language):
   216→    """Return (words, segments, duration, detected_language).
   217→
   218→    words: [{'word','start','end'}] with word timestamps from faster-whisper.
   219→    """
   220→    device, compute_type = pick_device()
   221→    size = resolve_model_size(model_size)
   222→    model = get_model(size, device, compute_type)
   223→    audio = load_audio_16k(audio_path)
   224→    lang = None if (not language or language == "auto") else language
   225→    segments_iter, info = model.transcribe(
   226→        audio,
   227→        language=lang,
   228→        word_timestamps=True,
   229→        condition_on_previous_text=False,
   230→        vad_filter=True,
   231→        beam_size=5,
   232→    )
   233→    words, segs = [], []
   234→    for seg in segments_iter:
   235→        segs.append({"start": seg.start, "end": seg.end, "text": seg.text.strip()})
   236→        for w in seg.words or []:
   237→            token = (w.word or "").strip()
   238→            if token:
   239→                words.append({"word": token, "start": float(w.start), "end": float(w.end)})
   240→    duration = float(getattr(info, "duration", 0.0)) or float(len(audio) / 16000.0)
   241→    detected = getattr(info, "language", None)
   242→    return words, segs, duration, detected, size

   250→def word_sim(a, b):
   251→    if not a or not b:
   252→        return 0.0
   253→    if a == b:
   254→        return 1.0
   255→    return SequenceMatcher(None, a, b).ratio()

   258→def dp_match(whisper_words, lyric_words):
   259→    """Monotonic alignment: lyric word j -> whisper word index or None.
   260→
   261→    Costs: match = 2 - sim (only if sim >= threshold); skip a heard word = 1.0;
   262→    leave a lyric word unmatched = 1.15.
   263→    """
   264→    n, m = len(whisper_words), len(lyric_words)
   265→    if n == 0 or m == 0:
   266→        return [None] * m
   267→    W = [normalize_word(w["word"]) for w in whisper_words]
   268→    L = [normalize_word(w) for w in lyric_words]
   269→    INF = float("inf")
   270→    SKIP_W, DEL_L = 1.0, 1.15
   271→    sim_ok = [[word_sim(W[i], L[j]) for j in range(m)] for i in range(n)]
   272→
   273→    # dp[i][j] = best cost of aligning W[i:] with L[j:]
   274→    dp = [[INF] * (m + 1) for _ in range(n + 1)]
   275→    dp[n][m] = 0.0
   276→    for i in range(n - 1, -1, -1):
   277→        dp[i][m] = dp[i + 1][m] + SKIP_W
   278→    for j in range(m - 1, -1, -1):
   279→        dp[n][j] = dp[n][j + 1] + DEL_L
   280→    for i in range(n - 1, -1, -1):
   281→        row, nxt = dp[i], dp[i + 1]
   282→        for j in range(m - 1, -1, -1):
   283→            best = nxt[j] + SKIP_W  # whisper word not used
   284→            if sim_ok[i][j] >= MATCH_SIM_THRESHOLD:
   285→                c = nxt[j + 1] + (2.0 - sim_ok[i][j])
   286→                if c < best:
   287→                    best = c
   288→            c2 = row[j + 1] + DEL_L  # lyric word left unmatched
   289→            if c2 < best:
   290→                best = c2
   291→            row[j] = best
   292→
   293→    mapping = [None] * m
   294→    i = j = 0
   295→    while i < n and j < m:
   296→        choices = []
   297→        if sim_ok[i][j] >= MATCH_SIM_THRESHOLD:
   298→            choices.append((dp[i + 1][j + 1] + (2.0 - sim_ok[i][j]), "match"))
   299→        choices.append((dp[i + 1][j] + SKIP_W, "skip"))
   300→        choices.append((dp[i][j + 1] + DEL_L, "del"))
   301→        choices.sort(key=lambda x: x[0])
   302→        _, choice = choices[0]
   303→        if choice == "match":
   304→            mapping[j] = i
   305→            i += 1
   306→            j += 1
   307→        elif choice == "skip":
   308→            i += 1
   309→        else:
   310→            j += 1
   311→    return mapping

   314→def assign_word_times(whisper_words, sing_entries, duration):
   315→    """Fill start/end for every word of every sing entry.
   316→
   317→    Model: matched words carry their Whisper times; unmatched words cling to
   318→    their OWN line's matched words (gaps live between lines, not inside a
   319→    line). Fully unheard lines are spread between the nearest heard anchors.
   320→    """
   321→    lyric_words = []
   322→    line_ranges = []
   323→    for e in sing_entries:
   324→        lo = len(lyric_words)
   325→        e["words"] = [{"text": w, "start": None, "end": None, "matched": False} for w in e["words"]]
   326→        lyric_words.extend(e["words"])
   327→        line_ranges.append((lo, len(lyric_words) - 1))
   328→    mapping = dp_match(whisper_words, [w["text"] for w in lyric_words])
   329→
   330→    total = len(lyric_words)
   331→    starts = [None] * total
   332→    ends = [None] * total
   333→    match_count = 0
   334→    for j, i in enumerate(mapping):
   335→        if i is not None:
   336→            starts[j] = whisper_words[i]["start"]
   337→            ends[j] = whisper_words[i]["end"]
   338→            match_count += 1
   339→
   340→    chars = [len(w["text"]) for w in lyric_words]
   341→    cum = [0.0]
   342→    for c in chars:
   343→        cum.append(cum[-1] + c + 1)
   344→    matched_js = [j for j in range(total) if starts[j] is not None]
   345→
   346→    if not matched_js:
   347→        # nothing matched at all: spread words evenly over the audio
   348→        span = max(duration - 1.0, 1.0)
   349→        for j in range(total):
   350→            starts[j] = 0.5 + span * j / max(total, 1)
   351→            ends[j] = starts[j] + 0.4
   352→    else:
   353→        # global median char rate as fallback
   354→        rates = []
   355→        for a, b in zip(matched_js, matched_js[1:]):
   356→            if cum[b] > cum[a]:
   357→                rates.append((starts[b] - starts[a]) / (cum[b] - cum[a]))
   358→        g_rate = min(0.25, max(0.04, sorted(rates)[len(rates) // 2])) if rates else 0.09
   359→
   360→        def line_rate(mjs):
   361→            if len(mjs) >= 2:
   362→                r = (starts[mjs[-1]] - starts[mjs[0]]) / max(cum[mjs[-1]] - cum[mjs[0]], 1e-6)
   363→                return min(0.30, max(0.03, r))
   364→            return g_rate
   365→
   366→        for lo, hi in line_ranges:
   367→            mjs = [j for j in range(lo, hi + 1) if starts[j] is not None]
   368→            if not mjs:
   369→                # whole line unheard: spread between nearest anchors outside it
   370→                prev = [j for j in matched_js if j < lo]
   371→                nxt = [j for j in matched_js if j > hi]
   372→                if prev and nxt:
   373→                    t0, t1 = ends[prev[-1]], starts[nxt[0]]
   374→                    span = max(0.0, t1 - t0)
   375→                    c0, c1 = cum[lo], cum[hi + 1]
   376→                    for j in range(lo, hi + 1):
   377→                        frac = (cum[j] - c0) / max(c1 - c0, 1e-6)
   378→                        starts[j] = t0 + span * frac
   379→                        ends[j] = starts[j] + chars[j] * g_rate
   380→                elif nxt:
   381→                    t = max(0.02, starts[nxt[0]] - (cum[nxt[0]] - cum[lo]) * g_rate)
   382→                    for j in range(lo, hi + 1):
   383→                        starts[j] = t
   384→                        ends[j] = t + chars[j] * g_rate
   385→                        t = ends[j] + 0.03
   386→                else:
   387→                    t = ends[prev[-1]] + 0.05
   388→                    for j in range(lo, hi + 1):
   389→                        starts[j] = t
   390→                        ends[j] = t + chars[j] * g_rate
   391→                        t = ends[j] + 0.03
   392→                continue
   393→            r = line_rate(mjs)
   394→            first_m, last_m = mjs[0], mjs[-1]
   395→            # leading unmatched words cling backwards to the first heard word
   396→            for j in range(first_m - 1, lo - 1, -1):
   397→                end_j = starts[j + 1] - 0.04
   398→                start_j = end_j - chars[j] * r
   399→                starts[j], ends[j] = start_j, max(start_j + 0.1, end_j)
   400→            # middle unmatched: char-weighted between same-line anchors
   401→            for a, b in zip(mjs, mjs[1:]):
   402→                if b - a <= 1:
   403→                    continue
   404→                base, span = ends[a], max(0.0, starts[b] - ends[a])
   405→                c0 = cum[a] + chars[a]
   406→                for j in range(a + 1, b):
   407→                    frac = (cum[j] + chars[j] / 2.0 - c0) / max(cum[b] - c0, 1e-6)
   408→                    frac = min(1.0, max(0.0, frac))
   409→                    center = base + span * frac
   410→                    half = chars[j] * r / 2.0
   411→                    starts[j], ends[j] = center - half, center + half
   412→            # trailing unmatched words cling forwards from the last heard word
   413→            prev_end = ends[last_m]
   414→            nxt_anchor = [j for j in matched_js if j > hi]
   415→            cap = starts[nxt_anchor[0]] - 0.05 if nxt_anchor else duration
   416→            for j in range(last_m + 1, hi + 1):
   417→                s = prev_end + 0.04
   418→                e = s + chars[j] * r
   419→                if e > cap:
   420→                    e = cap
   421→                    s = e - chars[j] * r
   422→                starts[j], ends[j] = s, max(e, s + 0.1)
   423→                prev_end = ends[j]
   424→
   425→    # sanity: finite + monotonic
   426→    for j in range(total):
   427→        if not np.isfinite(starts[j]):
   428→            starts[j] = 0.5
   429→        if not np.isfinite(ends[j]) or ends[j] <= starts[j]:
   430→            ends[j] = starts[j] + 0.3
   431→    for j in range(1, total):
   432→        if starts[j] < starts[j - 1]:
   433→            starts[j] = starts[j - 1]
   434→            ends[j] = max(ends[j], starts[j] + 0.2)
   435→
   436→    k = 0
   437→    for e in sing_entries:
   438→        for w in e["words"]:
   439→            w["start"], w["end"], w["matched"] = starts[k], ends[k], mapping[k] is not None
   440→            k += 1
   441→    return match_count, total

   444→def build_line_times(entries, duration):
   445→    """Line start = first word start, end = last word end (bounded by next line)."""
   446→    sing = [e for e in entries if e["kind"] == "sing"]
   447→    for e in sing:
   448→        if not e["words"]:
   449→            continue
   450→        e["start"] = e["words"][0]["start"]
   451→        e["end"] = max(e["words"][-1]["end"], e["start"] + MIN_LINE_DUR)
   452→    # clamp each line to the next sung line
   453→    for a in range(len(sing) - 1):
   454→        if sing[a + 1]["start"] > sing[a]["start"]:
   455→            sing[a]["end"] = min(sing[a]["end"], sing[a + 1]["start"] - 0.05)
   456→        sing[a]["end"] = max(sing[a]["end"], sing[a]["start"] + 0.3)
   457→    if sing:
   458→        sing[-1]["end"] = min(max(sing[-1]["end"], sing[-1]["start"] + 0.3), max(duration, sing[-1]["end"]))
   459→    # section tags take the time of the next sung line (or previous end)
   460→    last_end = 0.0
   461→    for e in entries:
   462→        if e["kind"] == "sing":
   463→            last_end = e["end"]
   464→        else:
   465→            nxt = next((s["start"] for s in sing if s["start"] > last_end - 1e-9), None)
   466→            e["start"] = nxt if nxt is not None else last_end
   467→            e["end"] = e["start"]

   475→def detect_beats(audio_16k):
   476→    try:
   477→        import librosa
   478→
   479→        tempo, beats = librosa.beat.beat_track(
   480→            y=audio_16k, sr=16000, units="time", trim=False
   481→        )
   482→        bpm = float(np.atleast_1d(tempo)[0])

   484→        return round(bpm, 1), beats
   485→    except Exception:
   486→        return None, []

   489→def build_cut_plan(sections, beats, bpm, duration, beats_per_clip):
   490→    """Beat-snapped clip slots. sections: [{'name','start','end'}]
   491→
   492→    Clip starts snap to the beat grid; a section's final slot absorbs any
   493→    short remainder instead of producing sub-second clips.
   494→    """
   495→    rows = []
   496→    if not sections:
   497→        return rows
   498→    if not beats or not bpm:
   499→        bpm = bpm or 120.0
   500→        step = beats_per_clip * 60.0 / bpm
   501→        beats = list(np.arange(0, duration, step))
   502→    nominal = beats_per_clip * 60.0 / bpm
   503→    n = 0
   504→    for sec in sections:
   505→        s0, s1 = sec["start"], min(sec["end"], duration)
   506→        if s1 - s0 < 1.0:
   507→            continue
   508→        # first beat at/after section start
   509→        idx = [i for i, b in enumerate(beats) if b >= s0 - 1e-6]
   510→        if not idx:
   511→            continue
   512→        i = idx[0]
   513→        sec_rows = []
   514→        while i < len(beats) and beats[i] < s1 - 0.4:
   515→            start = max(beats[i], s0)
   516→            # end = beats_per_clip beats later, or the section end
   517→            if i + beats_per_clip < len(beats):
   518→                end = min(beats[i + beats_per_clip], s1)
   519→            else:
   520→                end = s1
   521→            end = max(end, start + 0.5)
   522→            sec_rows.append({"start": start, "end": end})
   523→            i += beats_per_clip
   524→        # absorb a short trailing remainder into the previous slot
   525→        while len(sec_rows) >= 2 and sec_rows[-1]["end"] - sec_rows[-1]["start"] < nominal * 0.6:
   526→            last = sec_rows.pop()
   527→            sec_rows[-1]["end"] = last["end"]
   528→        for r in sec_rows:
   529→            n += 1
   530→            dur = r["end"] - r["start"]
   531→            rows.append(
   532→                {
   533→                    "clip": n,
   534→                    "section": sec["name"],
   535→                    "start": r["start"],
   536→                    "end": r["end"],
   537→                    "duration": round(dur, 2),
   538→                    "frames": int(round(dur * VIDEO_FPS)),
   539→                }
   540→            )
   541→    return rows

   544→def _collect_tags(entries, duration):
   545→    """Tag entries as [(name, time)], with a proper 0:00 intro chapter."""
   546→    tags = [(e["text"], float(e["start"])) for e in entries if e["kind"] == "tag"]
   547→    if tags and tags[0][1] > 1.0:
   548→        if "intro" in tags[0][0].strip().lower():
   549→            tags[0] = (tags[0][0], 0.0)
   550→        else:
   551→            tags.insert(0, ("Intro", 0.0))
   552→    return tags

   555→def build_sections(entries, duration):
   556→    """Sections from tag entries (YouTube chapters drive the spans)."""
   557→    tags = _collect_tags(entries, duration)
   558→    if not tags:
   559→        return [{"name": "Full song", "start": 0.0, "end": duration}]
   560→    sections = []
   561→    for k, (name, t0) in enumerate(tags):
   562→        t_end = tags[k + 1][1] if k + 1 < len(tags) else duration
   563→        sections.append({"name": name.title(), "start": min(t0, duration), "end": max(t_end, min(t0, duration) + 1.0)})
   564→    return sections

   567→def build_chapters(entries, duration):
   568→    tags = _collect_tags(entries, duration)
   569→    if not tags:
   570→        return [], "no [Verse]/[Chorus] tags found — add tags like [Chorus] to your lyrics to get YouTube chapters"
   571→    chapters = []
   572→    for k, (name, t) in enumerate(tags):
   573→        nxt = tags[k + 1][1] if k + 1 < len(tags) else duration
   574→        chapters.append({"name": name.title(), "start": t, "end": nxt})
   575→    notes = []
   576→    if len(chapters) < 3:
   577→        notes.append("YouTube needs at least 3 chapters")
   578→    if chapters and chapters[0]["start"] > 0.5:
   579→        notes.append("first chapter must start at 0:00 for YouTube")
   580→    short = [c["name"] for c in chapters if c["end"] - c["start"] < 10.0]
   581→    if short:
   582→        notes.append("chapters under 10 s: " + ", ".join(short))
   583→    return chapters, ("; ".join(notes) if notes else "valid for YouTube")

   613→    sing = [e for e in entries if e["kind"] == "sing"]

   644→def fmt_reaper(entries, duration):
   645→    """REAPER marker/region CSV: lyric lines -> markers, section tags -> regions.
   646→
   647→    Header columns are matched by name on import (REAPER 7.66+; drag the CSV into
   648→    the Region/Marker Manager). Times are H:MM:SS.mmm, one of the formats REAPER
   649→    parses natively.
   650→    """
   651→    rows = ["#,Name,Start,End,Length,Color"]
   652→    m_id = r_id = 0
   653→    sec = [e for e in entries if e["kind"] == "tag"]
   654→    sing = [e for e in entries if e["kind"] == "sing"]
   655→    for e in sing:
   656→        m_id += 1
   657→        rows.append(
   658→            f"M{m_id},{_csv_field(e['text'])},{fmt_clock(e['start'], force_hours=True)},,,"
   659→        )
   660→    for k, e in enumerate(sec):
   661→        r_id += 1
   662→        end = sec[k + 1]["start"] if k + 1 < len(sec) else max(duration, e["start"] + 1.0)
   663→        rows.append(
   664→            f"R{r_id},{_csv_field(e['text'].title())},{fmt_clock(e['start'], force_hours=True)},"
   665→            f"{fmt_clock(end, force_hours=True)},{fmt_clock(end - e['start'], force_hours=True)},"
   666→        )
   667→    return "\n".join(rows)

   756→def run_alignment(audio_path, lyrics_text, model_size="auto", language="auto", isolate_vocals_flag=False):
   757→    result = {
   758→        "song": os.path.splitext(os.path.basename(audio_path))[0],
   759→        "entries": [],
   760→        "chapters": [],
   761→        "chapter_note": "",
   762→        "beats": [],
   763→        "bpm": None,
   764→        "cut_plan": [],
   765→        "match_rate": None,
   766→        "status": "",
   767→        "audio_16k": None,
   768→        "duration": 0.0,
   769→    }
   770→    if not audio_path:
   771→        result["status"] = "⚠️ **Upload a song first** (step 1)."
   772→        return result
   773→
   774→    notes = []
   775→    src = audio_path
   776→    if isolate_vocals_flag:
   777→        try:
   778→            src = isolate_vocals(audio_path)
   779→            notes.append("🎛️ vocal isolation applied")
   780→        except Exception as exc:  # graceful fallback
   781→            notes.append(f"⚠️ vocal isolation unavailable ({exc} — continuing with the full mix)")
   782→
   783→    words, segs, duration, detected, size_used = transcribe_words(src, model_size, language)
   784→    result["duration"] = duration
   785→    result["detected_language"] = detected
   786→    result["model_used"] = size_used
   787→    result["device"] = pick_device()[0]
   788→
   789→    entries = parse_lyrics(lyrics_text)
   790→    sing = [e for e in entries if e["kind"] == "sing"]
   791→
   792→    if not entries:
   793→        # auto-transcribe mode: whisper segments become the lines, split into
   794→        # phrase-level rows (whisper merges sung lines; punctuation splits them)
   795→        entries = []
   796→        for s in segs:
   797→            text = (s["text"] or "").strip()
   798→            if not text:
   799→                continue
   800→            parts = [p.strip() for p in re.split(r"[,.;:!?—\-]+", text) if p.strip()]
   801→            if not parts:
   802→                parts = [text]
   803→            seg_dur = max(0.3, s["end"] - s["start"])
   804→            total_chars = sum(len(p) for p in parts)
   805→            t = s["start"]
   806→            for p in parts:
   807→                d = seg_dur * len(p) / max(total_chars, 1)
   808→                entries.append({"kind": "sing", "text": p, "start": t, "end": t + d, "words": []})
   809→                t += d
   810→        result["entries"] = entries
   811→        result["match_rate"] = None
   812→        result["status"] = (
   813→            f"✅ **Auto-transcribed** {len(entries)} lines with Whisper ({size_used} on {result['device']}). "
   814→            "You did not supply lyrics, so the text is what the AI heard — check it, or paste your real lyrics for guaranteed-correct text."
   815→        )
   816→    elif not sing:
   817→        result["entries"] = []
   818→        result["status"] = "⚠️ Your lyric sheet has no sung lines — only [tags]. Add the actual lyric lines."
   819→        return result
   820→    else:
   821→        match_count, total = assign_word_times(words, sing, duration)
   822→        build_line_times(entries, duration)
   823→        result["entries"] = entries
   824→        rate = match_count / max(total, 1)
   825→        result["match_rate"] = round(rate, 3)
   826→        pct = int(round(rate * 100))
   827→        if rate >= 0.75:
   828→            mood = "🟢"
   829→            advice = "Strong match — timings should be tight."
   830→        elif rate >= 0.55:
   831→            mood = "🟡"
   832→            advice = "Decent match — audition a few lines; try the offset slider if everything feels early/late."
   833→        else:
   834→            mood = "🔴"
   835→            advice = (
   836→                "Low match — Whisper struggled to hear your words. Try: **vocal isolation ON**, a **bigger model**, "
   837→                "or check **Language**. Also make sure the pasted lyrics match what is actually sung."
   838→            )
   839→        result["status"] = (
   840→            f"{mood} **{pct}% of your lyric words were located in the audio** "
   841→            f"({match_count}/{total}) — model `{size_used}` on {result['device']}, "
   842→            f"language heard: `{detected}`. {advice}"
   843→        )
   844→        if notes:
   845→            result["status"] += "\n\n" + " · ".join(notes)
   846→
   847→    # structure: chapters + sections + beats + cut plan
   848→    chapters, chapter_note = build_chapters(result["entries"], duration)
   849→    result["chapters"], result["chapter_note"] = chapters, chapter_note
   850→    sections = build_sections(result["entries"], duration)
   851→    audio16 = load_audio_16k(audio_path)
   852→    result["audio_16k"] = audio16
   853→    bpm, beats = detect_beats(audio16)
   854→    result["bpm"], result["beats"] = bpm, beats
   855→    result["cut_plan"] = build_cut_plan(sections, beats, bpm, duration, DEFAULT_BEATS_PER_CLIP)
   856→    result["sections"] = sections
   857→
   858→    tail = []
   859→    if chapters:
   860→        tail.append(f"📌 chapters: {chapter_note}")
   861→    elif chapter_note:
   862→        tail.append(f"📌 {chapter_note}")
   863→    if bpm:
   864→        tail.append(f"🥁 detected ≈ **{bpm} BPM**, {len(beats)} beats → Cut Plan uses 4-beat clips (~{4 * 60 / bpm:.1f} s)")
   865→    else:
   866→        tail.append("🥁 no beat grid detected — Cut Plan fell back to 2-second slots")
   867→    result["status"] += "\n\n" + " · ".join(tail)
   868→    return result

   885→        mark = "🏷️" if e["kind"] == "tag" else "🎵"

   890→            conf = "auto"

   892→            hit = sum(1 for w in words if w.get("matched"))
   893→            pct = int(round(100.0 * hit / len(words)))
   894→            conf = f"{pct}% — solid" if pct >= 85 else (f"{pct}% — ok" if pct >= 60 else f"{pct}% — check")
   895→        preview.append([i, fmt_plain(e["start"]), fmt_plain(e.get("end", e["start"])), f'{mark} {e["text"]}', conf])

   903→        "cutplan": fmt_cutplan(cut_rows),

   968→def start_cloudflare(port, timeout=90):
   969→    """Optional Cloudflare link (…trycloudflare.com) — same style as the YuE2 / H3 apps.
   970→
   971→    Downloads the checksum-pinned cloudflared binary, tunnels the local Gradio
   972→    port and returns the public URL. Any failure raises; the caller falls back
   973→    to the gradio.live share link, which works on its own.
   974→    """
   975→    import hashlib
   976→    import platform
   977→    import shutil
   978→    import subprocess
   979→    import time
   980→    import urllib.request
   981→    from pathlib import Path
   982→
   983→    if platform.system() != "Linux" or platform.machine().lower() not in ("x86_64", "amd64"):
   984→        raise RuntimeError("needs Linux x86_64 (Kaggle notebooks are fine)")
   985→    root = Path(tempfile.gettempdir()) / "sts_cloudflared"
   986→    root.mkdir(parents=True, exist_ok=True)
   987→    binary = root / "cloudflared"
   988→
   989→    def _digest(p):
   990→        h = hashlib.sha256()
   991→        with open(p, "rb") as fh:
   992→            for block in iter(lambda: fh.read(8 * 1024 * 1024), b""):
   993→                h.update(block)
   994→        return h.hexdigest()
   995→
   996→    if not binary.is_file() or _digest(binary) != CF_SHA256:
   997→        tmp = root / "cloudflared.download"
   998→        print("  Fetching the Cloudflare helper (~25 MB, one-time per session)…", flush=True)
   999→        with urllib.request.urlopen(CF_URL, timeout=60) as resp, open(tmp, "wb") as out:
  1000→            shutil.copyfileobj(resp, out)
  1001→        if _digest(tmp) != CF_SHA256:
  1002→            raise RuntimeError("cloudflared checksum mismatch")
  1003→        tmp.chmod(0o700)
  1004→        os.replace(tmp, binary)
